cmake_minimum_required(VERSION 3.21)
project(ItemSourceConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Quick Qml Test)
if(QM_CONSUMER_SOURCE)
  set(QM_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
  set(QM_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(QML_MATERIAL_BUILD_TYPE STATIC CACHE STRING "" FORCE)
  add_subdirectory("${QM_CONSUMER_SOURCE}" qml-material EXCLUDE_FROM_ALL)
else()
  find_package(qml_material CONFIG REQUIRED)
endif()
qt_add_executable(item_source_consumer main.cpp)
qt_add_qml_module(item_source_consumer URI Consumer VERSION 1.0 SOURCES track.hpp)
target_link_libraries(item_source_consumer PRIVATE qml_material::qml_material Qt6::Test)
if(QM_CONSUMER_SOURCE)
  target_link_libraries(item_source_consumer PRIVATE qml_materialplugin qml_material_layoutsplugin)
endif()
qt_import_qml_plugins(item_source_consumer)
